    const runtime = window.OuroborosMetadataEditorRuntime;
    const host = window.OuroborosMetadataEditorHost;
    if (!runtime || !host) {
      const target = document.getElementById('metadata-amis-root') || document.body;
      target.textContent = 'Metadata editor runtime failed to load.';
      throw new Error('Ouroboros metadata editor runtime or host failed to load');
    }
    Promise.resolve(typeof host.initialize === 'function' ? host.initialize() : host.initialState)
      .then(initialState => {
        const editorApp = runtime.mount({
          initialState,
          theme: host.theme,
          locale: host.locale || 'zh-CN',
          amisEditorDragFallback: host.amisEditorDragFallback,
          transport: host.transport || {}
        });
        window.OuroborosMetadataEditorApp = editorApp;
        if (typeof host.connect === 'function') {
          host.connect(editorApp);
        }
      })
      .catch(error => {
        const target = document.getElementById('metadata-amis-root') || document.body;
        target.textContent = error && error.message ? error.message : String(error);
      });
